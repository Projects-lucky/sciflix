import { trendingService, personService } from "@/services";
// Adjust this import path to match your actual transformer file location
import { transformTrendingResponse } from "@/transformers";
import { MediaGrid } from "@/components/shared/MediaGrid";
import { PersonGrid } from "@/components/person/PersonGrid";
import { HeroCarousel } from "@/components";

export const revalidate = 3600; // Revalidate cache every hour

export default async function HomePage() {
  // 1. FETCH: Concurrently fetch all required data streams
  const [
    trendingAllResult,
    trendingMoviesResult,
    trendingTvResult,
    popularPeopleResult,
  ] = await Promise.allSettled([
    trendingService.getTrending("all", "week"),
    trendingService.getTrendingMovies("week"),
    trendingService.getTrendingTV("week"),
    personService.getPopular(),
  ]);

  // 2. EXTRACT & TRANSFORM: Safely extract raw data and apply your existing transformer
  // We use a limit of 10 for the homepage grids to keep performance optimal
  const trendingAllData =
    trendingAllResult.status === "fulfilled"
      ? transformTrendingResponse(trendingAllResult.value, 10)
      : { results: [] };

  const trendingMoviesData =
    trendingMoviesResult.status === "fulfilled"
      ? transformTrendingResponse(trendingMoviesResult.value, 10)
      : { results: [] };

  const trendingTvData =
    trendingTvResult.status === "fulfilled"
      ? transformTrendingResponse(trendingTvResult.value, 10)
      : { results: [] };

  const popularPeopleData =
    popularPeopleResult.status === "fulfilled"
      ? transformTrendingResponse(popularPeopleResult.value, 10)
      : { results: [] };

  return (
    <div>
      <main className="mx-auto px-4 py-8">
        <h1 className="text-3xl font-bold mb-8">Welcome to MovieApp</h1>

        {/* 1. HERO CAROUSEL: Pass the transformed trending 'all' data */}
        <section>
          <HeroCarousel items={trendingAllData.results} />
        </section>
      </main>
    </div>
  );
}

'use client'

import Image from 'next/image'
import Link from 'next/link'
import { Calendar, MapPin, Star, ChevronLeft, Film, Tv } from 'lucide-react'
import type { PersonDetails, FilmographyItem } from '@/types'
import { TMDB_CONFIG } from "@/constants/api"

interface PersonDetailProps {
  person: PersonDetails
  filmography?: FilmographyItem[]
}

// DRY Helper: Centralized date formatting
const formatDate = (dateString: string) => 
  new Date(dateString).toLocaleDateString('en-US', { year: 'numeric', month: 'long', day: 'numeric' })

export function PersonDetail({ person, filmography }: PersonDetailProps) {
  // 1. Safely construct base URL
  const imageBase = TMDB_CONFIG.IMAGE_BASE_URL.endsWith('/') 
    ? TMDB_CONFIG.IMAGE_BASE_URL 
    : `${TMDB_CONFIG.IMAGE_BASE_URL}/`
  
  // 2. Use 'profile' as per your fix (with fallback to profile_path for absolute safety)
  const profilePath = person.profile || person.profile_path
  const profileSize = TMDB_CONFIG.IMAGE_SIZES?.profile?.medium || 'w342'
  const profileUrl = profilePath ? `${imageBase}${profileSize}${profilePath}` : '/placeholder-profile.jpg'

  // 3. DRY Metadata Configuration
  const metadataItems = [
    person.birthday && { icon: Calendar, label: 'Born', value: formatDate(person.birthday) },
    person.place_of_birth && { icon: MapPin, label: 'Place', value: person.place_of_birth },
    person.deathday && { icon: Calendar, label: 'Died', value: formatDate(person.deathday), className: 'text-red-500 dark:text-red-400' },
    { icon: Star, label: 'Popularity', value: person.popularity.toFixed(1), className: 'text-yellow-500 dark:text-yellow-400 font-semibold' }
  ].filter(Boolean) as { icon: any, label: string, value: string, className?: string }[]

  return (
    <div className="container mx-auto px-4 py-8 max-w-7xl">
      {/* Navigation */}
      <Link
        href="/people"
        className="inline-flex items-center gap-2 text-sm font-medium text-gray-500 hover:text-blue-600 dark:text-gray-400 dark:hover:text-blue-400 transition-colors mb-8 group"
      >
        <ChevronLeft className="w-4 h-4 transition-transform group-hover:-translate-x-1" />
        Back to People
      </Link>

      <div className="grid md:grid-cols-12 gap-8 lg:gap-12">
        {/* Profile Image (Sticky on Desktop for better UX) */}
        <div className="md:col-span-4 lg:col-span-3">
          <div className="sticky top-24 self-start">
            <div className="relative aspect-[2/3] rounded-2xl overflow-hidden shadow-xl bg-neutral-100 dark:bg-neutral-800 ring-1 ring-black/5 dark:ring-white/10">
              <Image
                src={profileUrl}
                alt={person.name}
                fill
                sizes="(max-width: 768px) 100vw, 33vw"
                className="object-cover"
                priority={true} // 🚀 LCP Optimization: Load main image immediately
              />
            </div>
            
            {/* Quick Stats Mobile Only (if needed, or keep in main flow) */}
          </div>
        </div>

        {/* Details Section */}
        <div className="md:col-span-8 lg:col-span-9 space-y-8">
          <div>
            <h1 className="text-4xl md:text-5xl font-extrabold tracking-tight text-gray-900 dark:text-white text-balance">
              {person.name}
            </h1>
            <p className="mt-2 text-lg font-medium text-blue-600 dark:text-blue-400">
              {person.known_for_department}
            </p>
          </div>
          
          {/* DRY Metadata Grid */}
          <div className="grid grid-cols-1 sm:grid-cols-2 gap-x-8 gap-y-4 py-6 border-y border-gray-100 dark:border-gray-800">
            {metadataItems.map((item, idx) => (
              <div key={idx} className="flex items-start gap-3">
                <item.icon className={`w-5 h-5 mt-0.5 shrink-0 text-gray-400 dark:text-gray-500`} />
                <div>
                  <span className="block text-xs font-semibold uppercase tracking-wider text-gray-400 dark:text-gray-500">
                    {item.label}
                  </span>
                  <span className={`text-sm md:text-base ${item.className || 'text-gray-700 dark:text-gray-200'}`}>
                    {item.value}
                  </span>
                </div>
              </div>
            ))}
          </div>

          {/* Biography */}
          {person.biography && (
            <div className="prose prose-neutral dark:prose-invert max-w-none">
              <h2 className="text-2xl font-bold mb-4 flex items-center gap-2">
                Biography
              </h2>
              <p className="text-gray-600 dark:text-gray-300 leading-relaxed text-balance">
                {person.biography}
              </p>
            </div>
          )}
        </div>
      </div>

      {/* Filmography Section */}
      {filmography && filmography.length > 0 && (
        <div className="mt-16 pt-8 border-t border-gray-100 dark:border-gray-800">
          <h2 className="text-2xl font-bold mb-6 flex items-center gap-2">
            <Film className="w-6 h-6 text-gray-400" />
            Known For
          </h2>
          
          <div className="grid grid-cols-2 sm:grid-cols-3 md:grid-cols-4 lg:grid-cols-5 xl:grid-cols-6 gap-4">
            {filmography.slice(0, 30).map((item) => {
              const posterSize = TMDB_CONFIG.IMAGE_SIZES?.poster?.medium || 'w342'
              const itemPosterUrl = item.poster ? `${imageBase}${posterSize}${item.poster}` : null
              const isMovie = item.mediaType === 'movie'

              return (
                <Link
                  key={`${item.id}-${item.mediaType}`}
                  href={isMovie ? `/movies/${item.id}` : `/tv/${item.id}`}
                  className="group block relative"
                >
                  <div className="relative aspect-[2/3] rounded-xl overflow-hidden bg-gray-100 dark:bg-gray-800 shadow-sm ring-1 ring-black/5 dark:ring-white/10 transition-all duration-300 group-hover:shadow-lg group-hover:ring-blue-500/30">
                    {itemPosterUrl ? (
                      <Image
                        src={itemPosterUrl}
                        alt={item.title}
                        fill
                        sizes="(max-width: 640px) 50vw, (max-width: 1024px) 33vw, 16vw"
                        className="object-cover transition-transform duration-500 group-hover:scale-105"
                        loading="lazy" // 🚀 Performance: Lazy load grid images
                      />
                    ) : (
                      <div className="absolute inset-0 flex flex-col items-center justify-center text-gray-400 dark:text-gray-600">
                        {isMovie ? <Film className="w-8 h-8 mb-2 opacity-50" /> : <Tv className="w-8 h-8 mb-2 opacity-50" />}
                        <span className="text-[10px] uppercase tracking-wider">No Image</span>
                      </div>
                    )}
                    
                    {/* Gradient Overlay & Text */}
                    <div className="absolute inset-0 bg-gradient-to-t from-black/90 via-black/20 to-transparent opacity-80 group-hover:opacity-100 transition-opacity" />
                    <div className="absolute bottom-0 left-0 right-0 p-3">
                      <div className="flex items-center gap-1.5 mb-1">
                        <span className="px-1.5 py-0.5 rounded text-[9px] font-bold uppercase tracking-wider bg-white/20 text-white backdrop-blur-sm">
                          {isMovie ? 'Movie' : 'TV'}
                        </span>
                      </div>
                      <p className="text-sm font-semibold text-white truncate drop-shadow-md">
                        {item.title}
                      </p>
                      <p className="text-[11px] text-gray-300 truncate">
                        {item.year} • {item.character || item.job || 'Unknown Role'}
                      </p>
                    </div>
                  </div>
                </Link>
              )
            })}
          </div>
        </div>
      )}
    </div>
  )
}
